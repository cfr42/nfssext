\makeatletter
\AddToHook {selectfont} [.] {\lfc@aux}
\def\lfc@aux#1{%
  \edef\tempa{\expandafter\detokenize\expandafter{\expanded{\expandafter\expandafter\expandafter\noexpand\csname\f@encoding/\f@family/\f@series/\f@shape\endcsname}}}%
  \expandafter\lfc@auxi\expandafter{\tempa}%
}
\def\lfc@auxi#1{%
  \NEWLINE
  \NEWLINE
  \TYPE{[lfc font]: #1}
  \NEWLINE
  \NEWLINE
}
% David Carlisle: https://chat.stackexchange.com/transcript/message/69317522#69317522
\def\lfc@math@aux#1{%
  \count0=0
  \loop
    \NEWLINE
    \TYPE{[lfc]: math font \the\count0: \expandafter\meaning\the\textfont\count0}
    \expandafter\ifx\the\textfont\count0 \nullfont
    \else \advance\count0 1
    \repeat
  \NEWLINE
}
\makeatother
