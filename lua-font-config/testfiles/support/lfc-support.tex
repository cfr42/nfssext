\AddToHook {selectfont} [.] {\lfc@aux}
\def\lfc@aux#1{%
  \edef\tempa{\expandafter\detokenize\expandafter{\expanded{\expandafter\expandafter\expandafter\noexpand\csname\f@encoding/\f@family/\f@series/\f@shape\endcsname}}}%
  \expandafter\lfc@auxi\expandafter{\tempa}%
}
\def\lfc@auxi#1{%
  \NEWLINE
  \NEWLINE
  \TYPE{[lfc font]: #1}
  \NEWLINE
  \NEWLINE
}
\makeatother
